import copy
import io

import pathlib
import re

BASE = "c60c192b9088b5f21fc199d574de17e89e5b4c94"
REPO = "szTheory/muhtari"
REPORT = pathlib.Path(".planning/phases/27-headless-developer-trace/27-CI-EVIDENCE.md")
STORE = pathlib.Path("build/phase27-evidence")
KINDS = ("windows-x64", "windows-arm64", "linux-trace_cost")
SCOPE = "one draft validation PR; initial automatic CI only; no dispatch/rerun/merge/release"

def need(ok, why):
    if not ok:
        raise ValueError(why)

def text(cmd):
    return subprocess.check_output(cmd, text=True).strip()

def sha(value, size):
    return isinstance(value, str) and re.fullmatch("[0-9a-f]{%d}" % size, value) is not None

def inventory(body):
    need("| receipt | disposition | reviewed_source | built_source | run_url | attempt |" in body,
         "missing receipt table header")
    rows = {}
    for line in body.splitlines():
        cells = [x.strip() for x in line.strip().strip("|").split("|")]
        if not cells or cells[0] not in KINDS:
            continue
        need(len(cells) == 6 and cells[0] not in rows, "malformed/duplicate receipt row")
        kind, status, reviewed, built, url, attempt = cells
        need(status in ("pending", "failed", "candidate", "accepted"), "invalid disposition")
        need(reviewed == BASE, "missing/wrong reviewed source")
        need(built == "pending" or sha(built, 40), "missing/invalid built source")
        need(url == "pending" or re.fullmatch(r"https://github.com/szTheory/muhtari/actions/runs/[1-9][0-9]*", url),
             "missing/invalid run URL")
        need(attempt == "pending" or re.fullmatch(r"[1-9][0-9]*", attempt), "missing/invalid attempt")
        if status in ("candidate", "accepted"):
            need(built != "pending" and url != "pending" and attempt != "pending", "incomplete candidate identity")
        rows[kind] = dict(status=status, built=built, url=url, attempt=attempt)
    need(set(rows) == set(KINDS), "missing required Windows/Linux receipt row")
    return rows

def source(built):
    need(sha(built, 40), "invalid source SHA")
    subprocess.run(["git", "cat-file", "-e", built + "^{commit}"], check=True)
    subprocess.run(["git", "diff", "--quiet", BASE, built, "--", ".", ":(exclude).planning"], check=True)

def authority(body, publish=False):
    fields = {}
    for label in ("selection", "status", "source", "answer", "branch", "head", "date", "scope"):
        values = re.findall(r"^Authority " + label + r": (.+)$", body, re.M)
        need(len(values) == 1 and values[0].strip(), "missing/duplicate authority " + label)
        fields[label] = values[0].strip()
    committed = text(["git", "show", "HEAD:" + str(REPORT)])
    for label, value in fields.items():
        need("Authority " + label + ": " + value in committed.splitlines(), "authority answer is not committed")
    need(fields["scope"] == SCOPE, "incorrect authority scope")
    need(re.fullmatch(r"\d{4}-\d{2}-\d{2}", fields["date"]), "missing/invalid authority date")
    need(fields["branch"] == text(["git", "branch", "--show-current"]), "authority branch changed")

def measurement(receipt, built):
    need(receipt["producer"] == "tests/perf/trace.cmake" and receipt["source_sha"] == built,
         "wrong trace producer/source")
    need(isinstance(receipt["compiler"], str) and receipt["compiler"].strip(), "missing compiler identity")
    need(isinstance(receipt["valgrind"], str) and receipt["valgrind"].startswith("valgrind-"),
         "missing Valgrind identity")
    need(sha(receipt["fixture_sha256"], 64), "invalid fixture hash")
    need(all(sha(receipt["runner_sha256"][k], 64) for k in ("dual", "off")), "invalid runner identities")
    m = receipt["measurement"]
    need(m["workload_sha256"] == receipt["fixture_sha256"], "fixture/workload identity mismatch")
    for key in ("on_ir60", "on_ir360", "off_ir60", "off_ir360"):
        need(isinstance(m[key], list) and len(m[key]) == 2
             and all(type(x) is int and x > 0 for x in m[key]), "missing/nonpositive samples: " + key)
    on = [m["on_ir360"][i] - m["on_ir60"][i] for i in range(2)]
    off = [m["off_ir360"][i] - m["off_ir60"][i] for i in range(2)]
    need(on[0] == on[1] and off[0] == off[1] and min(on + off) > 0, "empty/unstable sample deltas")
    delta = on[0] - off[0]
    trunc = lambda n: (abs(n) // 300) * (-1 if n < 0 else 1)
    expected = dict(on_raw_delta=on[0], off_raw_delta=off[0], delta_raw=delta,
                    on_ir_frame=trunc(on[0]), off_ir_frame=trunc(off[0]), delta_ir_frame=trunc(delta),
                    threshold_raw=9600, repeats=2)
    need(all(type(m[k]) is int and m[k] == v for k, v in expected.items()), "inconsistent measurement arithmetic")
    need(m["status"] == "PASS" and abs(delta) <= 9600, "unrounded hooks-off threshold failed")
    for family in ("runner_bytes", "archive_bytes"):
        sizes = receipt["size_observation"][family]
        need(all(type(sizes[k]) is int for k in ("dual", "off", "delta"))
             and sizes["dual"] > 0 and sizes["off"] > 0
             and sizes["delta"] == sizes["dual"] - sizes["off"], "invalid size observation")

def path_contract(workflow, script):
    start = workflow.index("      - name: CPU-backed trace hooks-off cost gate\n")
    end = workflow.index("      - name: Instructions per frame\n", start)
    gate = workflow[start:end]
    paths = {}
    for key in ("RUN_ON", "RUN_OFF"):
        values = re.findall(r"^\s+" + key + r": (\$\{\{ runner[.]temp \}\}/[^\r\n]+)$", gate, re.M)
        need(len(values) == 1, "missing/ambiguous source-bound step env path: " + key)
        paths[key] = values[0].replace("${{ runner.temp }}", "/same-runner-temp")
        need('-D' + key + '="$' + key + '"' in gate, "step env not passed to measurement: " + key)
    need(paths["RUN_ON"] != paths["RUN_OFF"]
         and len(paths["RUN_ON"]) == len(paths["RUN_OFF"]), "source runner paths not distinct/equal-length")
    need("-DMODE=measure" in gate and "-P tests/perf/trace.cmake" in gate
         and "TRACE_CONTROL" not in gate, "wrong real-measurement invocation")
    function = script.split("function(trace_measure)", 1)[1].split("endfunction()", 1)[0]
    guards = (
        r'if\(RUN_ON STREQUAL RUN_OFF\)\s*trace_error\("RUN_ON and RUN_OFF must name distinct installed runners"\)\s*endif\(\)',
        r'string\(LENGTH "\$\{RUN_ON\}" on_path_length\)',
        r'string\(LENGTH "\$\{RUN_OFF\}" off_path_length\)',
        r'if\(NOT on_path_length EQUAL off_path_length\)\s*trace_error\("installed runner paths have unequal lengths"\)\s*endif\(\)',
    )
    need(all(re.search(pattern, function) for pattern in guards), "source measurement path rejection guard missing")

def artifact(run_id, run, name):
    listing = api("actions/runs/" + run_id + "/artifacts?per_page=100")

    need(a["workflow_run"]["id"] == int(run_id), "artifact belongs to another run")

    data = get_bytes("actions/artifacts/" + str(a["id"]) + "/zip")
    STORE.mkdir(parents=True, exist_ok=True)

def receipts(rows, accepted=False):
    for kind, row in rows.items():
        need(row["status"] == "accepted" if accepted else row["status"] in ("candidate", "accepted"),
