cmake_minimum_required(VERSION 3.25)
project(muhtari_state_compat C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
foreach(side old new)
  add_executable(compat-${side} "${ROOT}/tests/core/state_compat.c"
    "${ROOT}/tests/cart/images.c" "${ROOT}/runner/sha256.c")
  target_include_directories(compat-${side} PRIVATE
    "${${side}_INCLUDE}" "${ROOT}/tests/cart" "${ROOT}/runner")
  target_link_libraries(compat-${side} PRIVATE "${${side}_LIBRARY}")
  target_compile_definitions(compat-${side} PRIVATE EXPECTED_BEHAVIOUR=${${side}_BEHAVIOUR})
  if(MSVC)
    target_compile_definitions(compat-${side} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(compat-${side} PRIVATE /W4 /WX)
  else()
    target_compile_options(compat-${side} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
